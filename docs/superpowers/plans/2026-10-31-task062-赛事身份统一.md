```python
"""EN/JP 卡组内容身份（task 062，PRD v1.39 FR-9 续）。

规范化口径（EN 双通道共用）：
1. 抽取 (name, set, number, count)；number 统一 str，空 set/number 统一 None；
2. 按 (name, set, number) 聚合 count，再排序（None 按空串）；
3. canonical JSON + SHA-256[:16]，保留各自源前缀。
不把 archetype、section、数组顺序、decklist_id、CN card_id、env、fetched_at 纳入身份。
JP 同形：同 official_card_id 重复条目 count 聚合后排序哈希（card_id 即身份材料）。
mik deckId 已是内容实体 ID，不经本模块。
"""

from __future__ import annotations

import hashlib
from typing import Any, Iterable

from ptcgdb.scrapers.raw_store import canonical_json

# (name, set, number, count) 输入条目；set/number 可缺
CardEntry = tuple[Any, Any, Any, Any]


def _norm_text(raw: Any) -> str | None:
    """set/number 归一：None/空串 → None，其余 str()。"""
    if raw is None:
        return None
    text = str(raw)
    return text if text else None


def canonical_card_entries(entries: Iterable[CardEntry]) -> list[list[Any]]:
    """(name, set, number, count) 流 → 聚合 + 排序的 canonical 列表。

    返回 [[name, set, number, count], ...]：同 (name,set,number) count 相加；
    排序键 (name, set or "", number or "")，确定性。
    """
    agg: dict[tuple[str, str | None, str | None], int] = {}
    for name, set_code, number, count in entries:
        key = (str(name), _norm_text(set_code), _norm_text(number))
        agg[key] = agg.get(key, 0) + int(count)
    return [
        [name, set_code, number, count]
        for (name, set_code, number), count in sorted(
            agg.items(), key=lambda kv: (kv[0][0], kv[0][1] or "", kv[0][2] or "")
        )
    ]


def content_digest(canon: list[list[Any]]) -> str:
    """canonical JSON → SHA-256[:16]。"""
    return hashlib.sha256(canonical_json(canon).encode("utf-8")).hexdigest()[:16]


def make_en_deck_id(source: str, entries: Iterable[CardEntry]) -> str:
    """EN 内容哈希 deck_id：{source}:{digest}（limitless / limitless_site）。"""
    return f"{source}:{content_digest(canonical_card_entries(entries))}"


def make_jp_deck_id(source: str, pairs: Iterable[tuple[Any, Any]]) -> str:
    """JP 内容哈希 deck_id：同 official_card_id count 聚合后排序哈希。"""
    agg: dict[str, int] = {}
    for card_id, count in pairs:
        key = str(card_id)
        agg[key] = agg.get(key, 0) + int(count)
    canon = [[card_id, count] for card_id, count in sorted(agg.items())]
    return f"{source}:{content_digest(canon)}"
```

```python
"""出战身份（task 062，PRD v1.39 FR-9.6 续）。

- appearance_key：源侧出战键（库内审计用，UNIQUE(tournament_id, appearance_key)，
  不进导出/SDK）；teamName 不作主身份，队员数组顺序不影响键。
- appearance_id：公开稳定 ID = sha256(f"{tournament_id}|{appearance_key}")[:16]。
- 完全相同重复行映射同一 ID；同 key 不同事实（rank/record 等）→ AppearanceConflictError，
  调用方中止该赛事，不后写覆盖。
"""

from __future__ import annotations

import hashlib
from typing import Any, Iterable

APPEARANCE_FACT_FIELDS = (
    "deck_id",
    "rank",
    "points",
    "player_ref",
    "record_wins",
    "record_losses",
    "record_ties",
)


class AppearanceConflictError(RuntimeError):
    """同 appearance_key 出现不同事实：中止该赛事，不后写覆盖。"""


def make_appearance_id(tournament_id: str, appearance_key: str) -> str:
    """公开稳定 ID：sha256(f"{tournament_id}|{appearance_key}")[:16]。"""
    digest = hashlib.sha256(
        f"{tournament_id}|{appearance_key}".encode("utf-8")
    ).hexdigest()
    return digest[:16]


def mik_appearance_key(
    pin_codes: Iterable[str | None],
    deck_id: str,
    *,
    team_name: str | None = None,
    fallback: str | None = None,
) -> tuple[str, bool]:
    """mik 键材料：排序后全队 pinCode + deckId。

    返回 (key, degraded)：无 pinCode 时降级 teamName → fallback（条目序），
    degraded=True 由调用方记 warning（可审计降级）。deck_id 传源侧原始 deckId。
    """
    pins = sorted(p for p in pin_codes if p)
    if pins:
        return f"mik|pins:{','.join(pins)}|deck:{deck_id}", False
    if team_name:
        return f"mik|team:{team_name}|deck:{deck_id}", True
    return f"mik|entry:{fallback}|deck:{deck_id}", True


def limitless_appearance_key(player: str | None, *, placing: int | None) -> str:
    """Limitless API 键材料：player；缺 player 降级 placing（可审计，调用方告警）。"""
    if player:
        return f"limitless|player:{player}"
    return f"limitless|placing:{placing}"


def site_appearance_key(player: str | None, decklist_id: str) -> str:
    """Limitless site 键材料：player + decklist_id。"""
    return f"site|player:{player}|decklist:{decklist_id}"


def jp_appearance_key(deck_code: str) -> str:
    """JP 键材料：官方 deck_code。"""
    return f"jp|code:{deck_code}"


def merge_appearance_facts(
    existing: dict[str, Any], new: dict[str, Any], *, context: str
) -> dict[str, Any]:
    """同 key 两条目：事实完全一致 → 去重返回 existing；任何字段冲突 → raise。"""
    for field in APPEARANCE_FACT_FIELDS:
        if existing.get(field) != new.get(field):
            raise AppearanceConflictError(
                f"同 appearance_key 冲突事实: {context} 字段 {field}: "
                f"{existing.get(field)!r} != {new.get(field)!r}"
            )
    return existing
```

- [ ] **Step 4: 改造 `ptcgdb/migrations/__init__.py`**

完整替换为：

```python
"""schema 迁移执行器（FR-6.4）。

`PRAGMA user_version` + `migrations/` 顺序 SQL 脚本，不用 Alembic。
重复执行幂等：user_version 记录已应用的最新版本，低于等于它的脚本跳过。
014 起支持 raw 驱动数据迁移（DATA_MIGRATIONS）：结构 SQL 以
`-- ===DATA-MIGRATION===` 分段，逐语句执行（executescript 会隐式提交，
破坏单事务），数据迁移夹在两段之间，同一事务成功后才置 user_version。
"""

from __future__ import annotations

import importlib
import re
import sqlite3
from pathlib import Path

MIGRATIONS_DIR = Path(__file__).parent
_FILENAME_RE = re.compile(r"^(\d+)_.*\.sql$")

SCHEMA_VERSION = "3.0.0"  # 结构版本 SemVer（FR-6.1）；破坏性变更升 major

# 版本号 → "module:func"：raw 驱动数据迁移钩子（task 062）
DATA_MIGRATIONS = {14: "ptcgdb.migrations.m014:run"}


class MigrationBlockedError(RuntimeError):
    """迁移前置条件不满足（如需 raw_dir 未提供）：库未变更，可补齐后重试。"""


def available_migrations() -> list[tuple[int, Path]]:
    """返回 [(版本号, SQL 文件路径)]，按版本号升序。"""
    migrations: list[tuple[int, Path]] = []
    for path in MIGRATIONS_DIR.glob("*.sql"):
        match = _FILENAME_RE.match(path.name)
        if match:
            migrations.append((int(match.group(1)), path))
    return sorted(migrations)


def _exec_statements(conn: sqlite3.Connection, script: str) -> None:
    """逐语句执行（保持当前事务；SQL 文件不含字符串内分号）。"""
    for stmt in script.split(";"):
        if stmt.strip():
            conn.execute(stmt)


def _apply_with_data_migration(
    conn: sqlite3.Connection, version: int, path: Path, raw_dir: Path | None
) -> None:
    """结构 SQL 分段 + raw 驱动数据迁移，同一事务；失败回滚不留半迁移。"""
    module_name, func_name = DATA_MIGRATIONS[version].split(":")
    run = getattr(importlib.import_module(module_name), func_name)
    script = path.read_text(encoding="utf-8")
    pre, sep, post = script.partition("-- ===DATA-MIGRATION===")
    if not sep:
        raise MigrationBlockedError(f"migration {version} 缺数据迁移分段标记: {path}")
    try:
        conn.execute("BEGIN IMMEDIATE")
        _exec_statements(conn, pre)
        run(conn, raw_dir)
        _exec_statements(conn, post)
        conn.execute(f"PRAGMA user_version = {version}")
        conn.execute("COMMIT")
    except Exception:
        conn.rollback()
        raise


def apply_migrations(
    db_path: str | Path,
    *,
    raw_dir: str | Path | None = None,
    up_to: int | None = None,
) -> int:
    """对 db_path 执行全部未应用的迁移，返回执行后的 user_version。

    raw_dir：014 等 raw 驱动数据迁移的原料目录；有数据的库缺省 →
    MigrationBlockedError（空库直迁不需要）。up_to：只迁到指定版本（测试用）。
    """
    db_path = Path(db_path)
    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(db_path)
    try:
        current = conn.execute("PRAGMA user_version").fetchone()[0]
        for version, path in available_migrations():
            if version <= current:
                continue
            if up_to is not None and version > up_to:
                break
            if version in DATA_MIGRATIONS:
                _apply_with_data_migration(
                    conn, version, path, Path(raw_dir) if raw_dir else None
                )
                continue
            script = path.read_text(encoding="utf-8")
            try:
                conn.executescript(
                    "BEGIN IMMEDIATE;\n"
                    + script
                    + f"\nPRAGMA user_version = {version};\nCOMMIT;"
                )
            except Exception:
                conn.rollback()
                raise
        # FR-6.1：新库写入、既有库升级到当前结构版本。
        conn.execute(
            "INSERT INTO meta (key, value) VALUES ('schema_version', ?) "
            "ON CONFLICT(key) DO UPDATE SET value=excluded.value",
            (SCHEMA_VERSION,),
        )
        conn.commit()
        return conn.execute("PRAGMA user_version").fetchone()[0]
    finally:
        conn.close()
```

注意：`PRAGMA user_version = N` 在事务内有效（SQLite user_version 随事务提交）。

        )

                ),
            )

"""共享 deck 映射环境裁决（task 062，PRD v1.39 FR-9.1b 续）。

ingest 与 remap 同一裁决：当前赛事与 DB 已有出战一起取最早非 NULL 日期 →
derive_env → allowed_marks；后抓到的更早赛事让裁决提前（触发重映射——
ingest 本就整组重写 deck_cards），较晚赛事不覆盖；完全无日期 → None（不猜）。
EN 双通道与 JP 一致执行；mik 卡 ID 直给不需要 env 映射（仅交叉校验，不经本模块）。
"""

from __future__ import annotations

from datetime import date
from typing import Any

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from ptcgdb.normalize.envs import SOURCE_REGION, derive_env
from ptcgdb.orm import DeckAppearance, Tournament


def earliest_deck_date(
    session: Session, deck_id: str, current_date: date | None = None
) -> date | None:
    """该 deck 最早非 NULL 出战日期：DB 已有出战 ∪ {current_date} 取最早。"""
    db_min = session.execute(
        select(func.min(Tournament.date)).where(
            Tournament.tournament_id.in_(
                select(DeckAppearance.tournament_id).where(
                    DeckAppearance.deck_id == deck_id
                )
            )
        )
    ).scalar_one_or_none()
    dates = [d for d in (db_min, current_date) if d is not None]
    return min(dates) if dates else None


def arbitrate_env_marks(
    session: Session,
    deck_id: str,
    source: str,
    current_date: date | None,
    env_calendar: dict[str, Any],
) -> tuple[str, ...] | None:
    """最早非 NULL 出战日期 ∩ 赛区日历段 → allowed_marks；无日期/未命中 → None。"""
    day = earliest_deck_date(session, deck_id, current_date)
    segment = derive_env(SOURCE_REGION.get(source), day, env_calendar)
    return segment.allowed_marks if segment is not None else None
```

def _scan_limitless(raw_dir: Path) -> tuple[dict[str, DeckBundle], list[AppearanceFact], set[str], dict[str, str]]:
    """EN API：standings → (bundles, appearances, seen_tournaments, old→new)。"""
    import hashlib

    bundles: dict[str, DeckBundle] = {}
    appearances: list[AppearanceFact] = []
    seen: set[str] = set()
    id_map: dict[str, str] = {}
    base = raw_dir / "limitless" / "tournaments"
    standings_dir = base / "standings"
    if not standings_dir.is_dir():
        return bundles, appearances, seen, id_map
    list_index = {
        tid: item for tid, (item) in _load_list_index(base, _NullResult()).items()
    }
    for path in sorted(standings_dir.glob("*.json")):
        doc = read_raw(path)
        if doc is None:
            continue
        tid = path.stem
        tournament_id = f"limitless:{tid}"
        seen.add(tournament_id)
        item = list_index.get(tid) or {}
        day = _parse_day(item.get("date"))
        fetched_at = _fetched_at(doc)
        by_key: dict[str, dict[str, Any]] = {}
        for e in doc.get("data") or []:
            if not isinstance(e, dict):
                continue
            entry = parse_standings_entry(e)
            if not entry.decklist_raw or not entry.decklist or entry.placing is None:
                continue
            old_id = "limitless:" + hashlib.sha256(
                canonical_json(entry.decklist_raw).encode("utf-8")
            ).hexdigest()[:16]
            new_id = make_en_deck_id(
                "limitless",
                ((c.name, c.set_code, c.number, c.count) for c in entry.decklist),
            )
            id_map[old_id] = new_id
            bundle = bundles.setdefault(
                new_id, DeckBundle(new_deck_id=new_id, source="limitless")
            )
            bundle.old_deck_ids.add(old_id)
            if not bundle.entries:
                bundle.entries = [
                    (c.name, c.set_code, c.number, c.count) for c in entry.decklist
                ]
            # 代表判定：最早出战（date 缺失排后，tournament_id/appearance_id 决胜）
            key = limitless_appearance_key(entry.player, placing=entry.placing)
            app_id = make_appearance_id(tournament_id, key)
            rep = (day or date.max, tournament_id, app_id)
            if bundle.rep_key is None or rep < bundle.rep_key:
                bundle.rep_key = rep
                bundle.archetype_id = entry.archetype_id
                bundle.archetype_name = entry.archetype_name
            if day:
                bundle.app_dates.append(day)
            facts = {
                "deck_id": new_id,
                "rank": entry.placing,
                "points": None,
                "player_ref": entry.player,
                "record_wins": entry.record_wins,
                "record_losses": entry.record_losses,
                "record_ties": entry.record_ties,
            }
            if key in by_key:
                merge_appearance_facts(
                    by_key[key], facts, context=f"{tournament_id} key={key}"
                )
            else:
                by_key[key] = facts
                appearances.append(
                    AppearanceFact(
                        appearance_id=app_id,
                        appearance_key=key,
                        tournament_id=tournament_id,
                        source="limitless",
                        fetched_at=fetched_at,
                        **facts,
                    )
                )
    return bundles, appearances, seen, id_map
```
